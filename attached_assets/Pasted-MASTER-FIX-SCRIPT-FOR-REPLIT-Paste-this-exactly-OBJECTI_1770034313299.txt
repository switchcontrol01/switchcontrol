type SystemInfo = {
  platform: string;
  arch: string;
  hostname: string;
  cpus: number;
  totalMemory: number;
  freeMemory: number;
  uptime: number;
};

useEffect(() => {
  if ((window as any).sc?.getSystemInfo) {
    (window as any).sc.getSystemInfo().then(setSystem);
  }
}, []);
