1. GPU vendor detection (Electron main)

You already have GPU info from systeminformation.

Add this once in main.js:

function isNvidiaGpu(graphics) {
  const gpu = graphics?.controllers?.[0];
  if (!gpu?.vendor) return false;
  return gpu.vendor.toLowerCase().includes('nvidia');
}

Store this in cached specs:

const gpuVendor = gpu?.vendor || 'Unknown';

cachedSpecs.gpu = {
  model: gpu?.model || 'Unavailable',
  vendor: gpuVendor,
  vramGB: gpu?.vram ? safeNum(gpu.vram / 1024) : 0,
  isNvidia: gpuVendor.toLowerCase().includes('nvidia')
};

This gives the frontend a truth flag to act on.

2. NVIDIA GPU temp (Electron main)

This is the only safe method.

Add helper (once)

const { exec } = require('child_process');

function getNvidiaGpuTemp() {
  return new Promise((resolve) => {
    exec(
      'nvidia-smi --query-gpu=temperature.gpu --format=csv,noheader,nounits',
      { windowsHide: true },
      (err, stdout) => {
        if (err) return resolve(null);
        const temp = Number(stdout.trim());
        resolve(Number.isFinite(temp) ? temp : null);
      }
    );
  });
}

Update telemetry:getLive

ipcMain.handle('telemetry:getLive', async () => {
  const load = await si.currentLoad();
  const cpuUsage = safeNum(load.currentLoad || 0);

  const total = os.totalmem();
  const free = os.freemem();
  const ramUsage = safeNum(((total - free) / total) * 100);

  let cpuTemp = 0;
  try {
    const temps = await si.cpuTemperature();
    cpuTemp = safeNum(temps.main || 0);
  } catch {}

  let gpuTemp = null;
  try {
    gpuTemp = await getNvidiaGpuTemp();
  } catch {}

  return {
    cpuUsage,
    ramUsage,
    cpuTemp: Number.isFinite(cpuTemp) ? cpuTemp : null,
    gpuTemp: Number.isFinite(gpuTemp) ? gpuTemp : null,
    timestamp: Date.now()
  };
});

Important:

null means unavailable

Never return undefined

Never fake values
