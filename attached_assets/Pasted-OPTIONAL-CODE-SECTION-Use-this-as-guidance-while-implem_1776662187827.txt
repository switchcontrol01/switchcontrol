Example updater state guards

function canCheck(status) {
  return ['idle', 'not-available', 'error', 'available'].includes(status);
}

function canDownload(status) {
  return status === 'available';
}

function canInstall(status) {
  return status === 'downloaded';
}

Example strict action guards

function checkForUpdates() {
  if (!_autoUpdater) {
    console.warn('[Updater] checkForUpdates called before init.');
    return;
  }
  if (!canCheck(state.status)) {
    console.warn('[Updater] checkForUpdates blocked in state:', state.status);
    return;
  }

  state = {
    ...state,
    status: 'checking',
    availableVersion: null,
    releaseNotes: null,
    releaseDate: null,
    errorMessage: null,
    downloadPercent: 0,
    bytesPerSecond: 0,
    transferred: 0,
    total: 0,
    checkedAt: new Date().toISOString(),
  };

  broadcast('checking-for-update');

  try {
    _autoUpdater.checkForUpdates();
  } catch (err) {
    const msg = err?.message || 'checkForUpdates failed';
    state = { ...state, status: 'error', errorMessage: msg };
    broadcast('error');
  }
}

function downloadUpdate() {
  if (!_autoUpdater) {
    console.warn('[Updater] downloadUpdate called before init.');
    return;
  }
  if (!canDownload(state.status)) {
    console.warn('[Updater] downloadUpdate blocked in state:', state.status);
    return;
  }

  try {
    _autoUpdater.downloadUpdate();
  } catch (err) {
    const msg = err?.message || 'downloadUpdate failed';
    state = { ...state, status: 'error', errorMessage: msg };
    broadcast('error');
  }
}

function quitAndInstall() {
  if (!_autoUpdater) {
    console.warn('[Updater] quitAndInstall called before init.');
    return;
  }
  if (!canInstall(state.status)) {
    console.warn('[Updater] quitAndInstall blocked in state:', state.status);
    return;
  }

  _autoUpdater.quitAndInstall(false, true);
}

Example version validation step in workflow

node -e "
const fs = require('fs');
const root = JSON.parse(fs.readFileSync('package.json', 'utf8'));
const electron = JSON.parse(fs.readFileSync('electron/package.json', 'utf8'));
if (root.version !== electron.version) {
  console.error('Version mismatch:', root.version, electron.version);
  process.exit(1);
}
console.log('Version sync OK:', root.version);
"
