File:

Problem:
Production must only allow switchcontrol.org domains.

const allowedOrigins = process.env.NODE_ENV === "production"
  ? [
      "https://switchcontrol.org",
      "https://www.switchcontrol.org"
    ]
  : [
      "http://localhost:3000",
      "http://127.0.0.1:3000"
    ];

app.use(
  cors({
    origin: function(origin, callback) {

      if (!origin) return callback(null, true);

      if (allowedOrigins.includes(origin)) {
        return callback(null, true);
      }

      return callback(new Error("CORS blocked"));
    },

    credentials: true
  })
);

Previously only logged warning.

Requirement:
Fail startup if secret < 32 bytes.

Implementation:

function getSecret() {

  const secret = process.env.JWT_SECRET;

  if (!secret) {
    throw new Error("JWT_SECRET missing");
  }

  if (process.env.NODE_ENV === "production" && secret.length < 32) {
    throw new Error("JWT_SECRET must be at least 32 characters in production");
  }

  return secret;
}

4. HARDEN STRUCTURED LOG REDACTION

File:
server/index.ts logging middleware

Replace current redaction logic with:

function sanitize(obj) {

  if (!obj) return obj;

  const redactedKeys = [
    "token",
    "authorization",
    "accessToken",
    "refreshToken",
    "jwt"
  ];

  const clone = { ...obj };

  for (const key of Object.keys(clone)) {
    if (redactedKeys.includes(key)) {
      clone[key] = "[REDACTED]";
    }
  }

  return clone;
}

Use this sanitizer before logging responses.

8. ADD POST-BUILD SECRET SCAN

Create script:

scripts/security-scan.js

Implementation:

const fs = require("fs");
const path = require("path");

const scanTargets = [
  "dist",
  "electron/dist"
];

const patterns = [
  "sk-",
  "JWT_SECRET",
  "DATABASE_URL",
  "OPENAI_API_KEY"
];

function scanDir(dir) {

  const files = fs.readdirSync(dir);

  for (const file of files) {

    const full = path.join(dir, file);

    if (fs.statSync(full).isDirectory()) {
      scanDir(full);
      continue;
    }

    const data = fs.readFileSync(full, "utf8");

    for (const p of patterns) {
      if (data.includes(p)) {
        console.warn("Possible secret found:", full);
      }
    }
  }
}

scanTargets.forEach(scanDir);

console.log("Security scan complete.");

Add npm script:

"scan:secrets": "node scripts/security-scan.js"
