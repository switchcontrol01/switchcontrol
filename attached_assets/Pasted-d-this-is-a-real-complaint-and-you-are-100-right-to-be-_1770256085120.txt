1. Add a PREBUILD SCRIPT that ENFORCES icon + folders

Create this file:

electron/scripts/ensure-assets.js

const fs = require("fs");
const path = require("path");

const buildDir = path.join(__dirname, "..", "build");
const iconPath = path.join(buildDir, "icon.ico");

if (!fs.existsSync(buildDir)) {
  fs.mkdirSync(buildDir, { recursive: true });
  console.log("[ensure-assets] Created electron/build/");
}

if (!fs.existsSync(iconPath)) {
  console.error("[ensure-assets] ERROR: electron/build/icon.ico is missing.");
  process.exit(1);
}

console.log("[ensure-assets] icon.ico present");

This script kills the build if the icon is missing. No lies. No silent fallback.


---

2. Force Electron to COPY frontend dist every time

Add this file:

electron/scripts/ensure-dist.js

const fs = require("fs");
const path = require("path");

const rootDist = path.join(__dirname, "..", "..", "dist");
const electronDist = path.join(__dirname, "..", "dist");

if (!fs.existsSync(rootDist)) {
  console.error("[ensure-dist] ERROR: root dist/ does not exist. Run npm run build first.");
  process.exit(1);
}

fs.rmSync(electronDist, { recursive: true, force: true });
fs.mkdirSync(electronDist, { recursive: true });

fs.cpSync(rootDist, electronDist, { recursive: true });

console.log("[ensure-dist] Frontend dist copied into electron/dist");

This guarantees Electron always has the frontend.
