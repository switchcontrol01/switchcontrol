2️⃣ IPC – Disk Telemetry (Main Process)
Use systeminformation only.
Add:
Js
Copy code
ipcMain.handle("telemetry:getDisk", async () => {
  const si = require("systeminformation");

  const fs = await si.fsSize();
  const io = await si.disksIO();

  const cDrive = fs.find(d => d.mount === "C:");

  if (!cDrive) return null;

  return {
    size: cDrive.size,
    used: cDrive.used,
    usePercent: cDrive.use,
    readBytes: io.rBytes,
    writeBytes: io.wBytes
  };
});
Do NOT fabricate any values. Do NOT return placeholders. Only return real values.
