import { useState, useEffect } from "react";
import { motion } from "framer-motion";
import { Button } from "@/components/ui/button";

const AUTH_BASE_URL = "https://switchcontrol.app";

interface AuthUser {
  id: string;
  email: string | null;
  name: string | null;
  avatar: string | null;
  isPremium: boolean;
}

interface LoginProps {
  onLoginSuccess: () => void;
}

declare global {
  interface Window {
    electronAPI?: {
      isElectron: boolean;
      system?: {
        openExternal: (url: string) => Promise<void>;
      };
      auth?: {
        onCallback: (cb: (data: { token: string; user: AuthUser }) => void) => void;
        removeCallbackListener: () => void;
      };
    };
  }
}

export default function Login({ onLoginSuccess }: LoginProps) {
  const [loading, setLoading] = useState<"google" | "discord" | null>(null);

  useEffect(() => {
    if (!window.electronAPI?.auth) return;

    const handler = (data: { token: string; user: AuthUser }) => {
      localStorage.setItem("sc_auth_token", data.token);
      localStorage.setItem("sc_auth_user", JSON.stringify(data.user));
      onLoginSuccess();
    };

    window.electronAPI.auth.onCallback(handler);
    return () => window.electronAPI?.auth?.removeCallbackListener();
  }, [onLoginSuccess]);

  const startAuth = async (provider: "google" | "discord") => {
    setLoading(provider);

    const url = `${AUTH_BASE_URL}/auth/${provider}?redirect=switchcontrol://auth`;

    if (window.electronAPI?.system?.openExternal) {
      await window.electronAPI.system.openExternal(url);
    }
  };

  return (
    <div className="fixed inset-0 flex items-center justify-center bg-[#0a0a0f] overflow-hidden">
      <div className="absolute inset-0 pointer-events-none rotate-[-12deg] scale-[1.4] opacity-10">
        <svg width="100%" height="100%">
          <defs>
            <pattern id="loginTopo" width="200" height="140" patternUnits="userSpaceOnUse">
              <path d="M0 40 Q50 10 100 40 T200 40" fill="none" stroke="hsl(270 50% 50%)" strokeWidth="0.6" />
              <path d="M0 90 Q50 60 100 90 T200 90" fill="none" stroke="hsl(275 45% 55%)" strokeWidth="0.5" />
            </pattern>
          </defs>
          <rect width="300%" height="300%" x="-100%" y="-100%" fill="url(#loginTopo)" />
        </svg>
      </div>

      <motion.div
        initial={{ opacity: 0, y: 16 }}
        animate={{ opacity: 1, y: 0 }}
        transition={{ duration: 0.4 }}
        className="relative z-10 w-full max-w-md px-6"
      >
        <div className="rounded-2xl border border-white/10 bg-card/80 backdrop-blur-xl p-8">
          <div className="flex flex-col items-center gap-6 mb-8">
            <img
              src="/logo.png"
              alt="SwitchControl"
              className="w-20 h-20 rounded-[20px] object-contain"
            />
            <h1 className="text-2xl font-bold text-white">Sign in</h1>
          </div>

          <div className="flex flex-col gap-3">
            <Button
              disabled={loading !== null}
              onClick={() => startAuth("google")}
            >
              Continue with Google
            </Button>

            <Button
              disabled={loading !== null}
              variant="secondary"
              onClick={() => startAuth("discord")}
            >
              Continue with Discord
            </Button>
          </div>
        </div>
      </motion.div>
    </div>
  );
}

const { app, BrowserWindow, ipcMain, shell } = require("electron");
const path = require("path");

let mainWindow;

function createWindow() {
  mainWindow = new BrowserWindow({
    width: 1200,
    height: 800,
    frame: false,
    webPreferences: {
      preload: path.join(__dirname, "preload.js"),
    },
  });

  mainWindow.loadURL(
    process.env.VITE_DEV_SERVER_URL ||
      `file://${path.join(__dirname, "../dist/index.html")}`
  );
}

app.whenReady().then(() => {
  app.setAsDefaultProtocolClient("switchcontrol");
  createWindow();
});

app.on("open-url", (event, url) => {
  event.preventDefault();

  const parsed = new URL(url);
  if (parsed.hostname !== "auth") return;

  const payload = {
    token: parsed.searchParams.get("token"),
    user: JSON.parse(parsed.searchParams.get("user") || "{}"),
  };

  mainWindow.webContents.send("auth:callback", payload);
});

ipcMain.handle("system:openExternal", async (_, url) => {
  await shell.openExternal(url);
});

const { contextBridge, ipcRenderer } = require("electron");

contextBridge.exposeInMainWorld("electronAPI", {
  isElectron: true,

  system: {
    openExternal: (url) => ipcRenderer.invoke("system:openExternal", url),
  },

  auth: {
    onCallback: (cb) => ipcRenderer.on("auth:callback", (_, data) => cb(data)),
    removeCallbackListener: () => ipcRenderer.removeAllListeners("auth:callback"),
  },
});
