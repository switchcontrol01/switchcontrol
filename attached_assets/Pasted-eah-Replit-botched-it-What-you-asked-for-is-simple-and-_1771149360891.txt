The actual implementation (this is what should be in your code)

1) Add a reusable hook: client/src/hooks/useRevealOnScroll.ts

import { useEffect, useRef } from "react";

type Options = {
  rootMargin?: string;
  threshold?: number;
};

export function useRevealOnScroll(options: Options = {}) {
  const ref = useRef<HTMLElement | null>(null);

  useEffect(() => {
    const el = ref.current;
    if (!el) return;

    const reduceMotion =
      typeof window !== "undefined" &&
      window.matchMedia &&
      window.matchMedia("(prefers-reduced-motion: reduce)").matches;

    if (reduceMotion) {
      el.classList.add("reveal-in");
      return;
    }

    const obs = new IntersectionObserver(
      (entries) => {
        for (const entry of entries) {
          if (entry.isIntersecting) {
            entry.target.classList.add("reveal-in");
            obs.unobserve(entry.target);
          }
        }
      },
      {
        root: null,
        rootMargin: options.rootMargin ?? "0px 0px -10% 0px",
        threshold: options.threshold ?? 0.15,
      }
    );

    obs.observe(el);

    return () => obs.disconnect();
  }, [options.rootMargin, options.threshold]);

  return ref;
}

2) Add the reveal CSS once (example: client/src/styles/reveal.css or wherever your global CSS is)

.reveal {
  opacity: 0;
  transform: translate3d(0, 18px, 0);
  filter: blur(10px);
  transition: opacity 700ms ease, transform 700ms ease, filter 700ms ease;
  will-change: opacity, transform, filter;
}

.reveal.reveal-in {
  opacity: 1;
  transform: translate3d(0, 0, 0);
  filter: blur(0px);
}

/* stagger support */
.reveal[data-delay="1"] { transition-delay: 60ms; }
.reveal[data-delay="2"] { transition-delay: 120ms; }
.reveal[data-delay="3"] { transition-delay: 180ms; }
.reveal[data-delay="4"] { transition-delay: 240ms; }

/* mobile performance guard */
@media (max-width: 768px) {
  .reveal {
    filter: blur(6px);
    transform: translate3d(0, 12px, 0);
    transition-duration: 600ms;
  }
}

/* accessibility */
@media (prefers-reduced-motion: reduce) {
  .reveal {
    transition: none;
    opacity: 1;
    transform: none;
    filter: none;
  }
}

Make sure this CSS is actually imported. If you use index.css or main.tsx, import it there.

3) Use it in your pages (Landing, Pricing)

Example usage in Landing.tsx:

import { useRevealOnScroll } from "../hooks/useRevealOnScroll";

function RevealSection(props: React.PropsWithChildren<{ delay?: number; className?: string }>) {
  const ref = useRevealOnScroll();
  return (
    <section
      ref={ref as any}
      className={`reveal ${props.className ?? ""}`}
      data-delay={props.delay ?? 0}
    >
      {props.children}
    </section>
  );
}

export default function Landing() {
  return (
    <div>
      <RevealSection className="hero">
        {/* hero content */}
      </RevealSection>

      <RevealSection delay={1}>
        {/* features */}
      </RevealSection>

      <RevealSection delay={2}>
        {/* pricing teaser */}
      </RevealSection>

      <RevealSection delay={3}>
        {/* CTA */}
      </RevealSection>
    </div>
  );
}

If you already have sections, just wrap them or add ref + className="reveal" directly.
