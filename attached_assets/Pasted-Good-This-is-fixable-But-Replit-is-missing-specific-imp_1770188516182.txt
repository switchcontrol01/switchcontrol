Backend (Node / Electron main)

Add this once and cache it:

import si from "systeminformation";

async function getGpuInfo() {
  const graphics = await si.graphics();
  const controller = graphics.controllers?.[0];

  if (!controller) {
    return {
      name: null,
      vramGB: null,
      vendor: null
    };
  }

  return {
    name: controller.model ?? controller.name ?? null,
    vramGB: controller.vram ? Math.round(controller.vram / 1024) : null,
    vendor: controller.vendor ?? null
  };
}
